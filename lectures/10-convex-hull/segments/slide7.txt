Phần 7. Code chi tiết Rotating Calipers (đường kính).
ll diameter(const vector<Point>& h) {
  int n = h.size(); if (n <= 1) return 0; if (n == 2) return dist2(h[0], h[1]);
  ll ans = 0; int j = 1;
  for (int i = 0; i < n; ++i) {
    int ni = (i+1)%n;
    while (ccw(h[i], h[ni], h[(j+1)%n]) > ccw(h[i], h[ni], h[j])) j = (j+1)%n;
    ans = max(ans, dist2(h[i], h[j]));
    ans = max(ans, dist2(h[ni], h[j]));
  }
  return ans;
}
dist2 = (x1-x2)^2 + (y1-y2)^2. Trả về bình phương khoảng cách.
