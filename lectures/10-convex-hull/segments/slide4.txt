vector<Point> convexHull(vector<Point> pts) {
  sort(pts.begin(), pts.end());
  pts.erase(unique(pts.begin(), pts.end()), pts.end()); // bỏ trùng
  int n = pts.size(); if (n <= 2) return pts;
  vector<Point> h(2*n); int k = 0;
  // Vỏ dưới
  for (int i = 0; i < n; ++i) {
    while (k >= 2 && ccw(h[k-2], h[k-1], pts[i]) <= 0) k--;
    h[k++] = pts[i];
  }
  // Vỏ trên
  for (int i = n-2, t = k+1; i >= 0; --i) {
    while (k >= t && ccw(h[k-2], h[k-1], pts[i]) <= 0) k--;
    h[k++] = pts[i];
  }
  h.resize(k-1);
  return h;
}
Điểm trùng: unique sau sort. n=1,2 trả về chính nó.
