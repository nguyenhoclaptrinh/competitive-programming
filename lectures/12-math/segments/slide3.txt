ll extgcd(ll a,ll b,ll&x,ll&y){if(b==0){x=1;y=0;return a;}ll x1,y1;ll d=extgcd(b,a%b,x1,y1);x=y1;y=x1-y1*(a/b);return d;}
bool crtMerge(ll r1,ll m1,ll r2,ll m2,ll&rem,ll&mod){
  ll x,y; ll g=extgcd(m1,m2,x,y);
  if((r2-r1)%g!=0) return false;
  ll step=m2/g; ll k=((r2-r1)/g%step)*(x%step)%step; if(k<0) k+=step;
  mod=m1/g*m2; rem=(r1+k*m1)%mod; if(rem<0) rem+=mod; return true;
}
